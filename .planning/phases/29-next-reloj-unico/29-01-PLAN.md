Forma del objeto (D-01, estricta — no agregar campos):
  lead.nextAction = {
    tipo,       // 'callback' | 'cadencia' | 'enviar_info' | 'esperar_respuesta' | 'otro'
    dueAt,      // string ISO (el MISMO string que se espeja a lead.callbackAt)
    canal,      // 'llamada' | 'whatsapp' | 'email' | ''
    motivo,     // string libre, <= 200 chars
    origen,     // 'manual' | 'cadencia' | 'compromiso'
    createdAt,  // ISO de cuándo se programó
    createdBy   // nombre del actor ('' si no se sabe)
  }
  lead.nextAction = null  → el lead no tiene próximo paso (D-01)

Helpers (firmas exactas):
  _setNextAction(lead, spec, nowIso) -> objeto escrito | null
  _clearNextAction(lead) -> void
  _deriveNextActionFromLegacy(lead) -> objeto | null      (PURA: no muta el lead)
  _leadNextAction(lead) -> objeto | null                  (PURA: nextAction o derivación)
  _nextActionTemplateForDelta(deltaMs) -> {key,label,deltaMs} | null
  NEXT_ACTION_TEMPLATES -> [{key:'24hs',label:'24h',deltaMs},{'48hs'},{'72hs'},{'7d'},{'15d'}]
