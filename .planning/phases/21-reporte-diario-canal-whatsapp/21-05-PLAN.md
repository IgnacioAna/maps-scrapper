// en DOS computadoras con la misma sesión, las dos reciben el comando y las dos

// Dedupe en memoria por queueId con TTL de 15 min.
const _reportSeen = new Map(); // queueId → ts
function _reportAlreadyHandled(queueId) {
  const now = Date.now();
  for (const [k, ts] of _reportSeen) if (now - ts > 15 * 60 * 1000) _reportSeen.delete(k);
  if (_reportSeen.has(queueId)) return true;

}

socket.on("report:send-message", async (p) => {

  const text = typeof p?.text === "string" ? p.text : "";
  const target = p?.target || {};
  const reply = (ok, method, reason, extra) => socket?.emit("report:send-result", {

  if (target.kind === "dm" && !/^\+?\d{8,15}$/.test(String(target.phone || "").replace(/[^\d+]/g, ""))) return reply(false, "dm", "exception");

| T-21-23 | Tampering | handler `report:send-message` | mitigate | validación estricta antes de ejecutar: `queueId` string ≤80, `text` no vacío y ≤4000 chars, `target.kind` en whitelist `{group,dm}`, `phone` contra `/^\+?\d{8,15}$/`. Cualquier payload fuera de contrato se descarta sin tipear |
